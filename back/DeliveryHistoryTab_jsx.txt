import React, { useState, useEffect } from 'react'
import { supabase } from '../../supabaseClient'

export default function ConsumablesTab({ partsList, refreshParts, showAddForm, setShowAddForm }) {
  const [searchTerm, setSearchTerm] = useState('')
  const [companies, setCompanies] = useState([])
  const [historyList, setHistoryList] = useState([]) // 💡 입·출고증 발행 이력 목록 상태

  // 1. 빠른 입고 로트 모달 상태
  const [quickLotPart, setQuickLotPart] = useState(null)
  const [quickLotNo, setQuickLotNo] = useState('')
  const [quickReceivedDate, setQuickReceivedDate] = useState(new Date().toISOString().split('T')[0])
  const [quickExpiry, setQuickExpiry] = useState('')
  const [quickQty, setQuickQty] = useState('')

  // 2. 소모품 전용 출고 모달 상태 (로트별 입력 필드 포함)
  const [outboundPart, setOutboundPart] = useState(null)
  const [outboundCompany, setOutboundCompany] = useState('')
  const [outboundDate, setOutboundDate] = useState(new Date().toISOString().split('T')[0])
  const [outboundNote, setOutboundNote] = useState('')
  const [showCompanyDropdown, setShowCompanyDropdown] = useState(false)
  const [outboundLotInputs, setOutboundLotInputs] = useState({}) // { lotIndex: 출고할수량 }

  // 3. 신규 소모품 등록 폼 상태
  const [partName, setPartName] = useState('')
  const [partCode, setPartCode] = useState('')
  const [partCategory, setPartCategory] = useState('잉크')
  const [dealerPrice, setDealerPrice] = useState(0)
  const [customerPrice, setCustomerPrice] = useState(0)
  const [partNote, setPartNote] = useState('')
  const [lots, setLots] = useState([])
  const [companyPrices, setCompanyPrices] = useState([])

  // 4. 유통기한 상세 확인 모달 상태
  const [selectedExpiryPart, setSelectedExpiryPart] = useState(null)

  // 5. 업체별 지정 특가 관리 모달 상태
  const [selectedSpecialPart, setSelectedSpecialPart] = useState(null)
  const [selectedSpecialPrices, setSelectedSpecialPrices] = useState(null)
  const [specialCompanySearchIndex, setSpecialCompanySearchIndex] = useState(null)

  // 6. 품목 정보 수정 모달 상태
  const [editingPart, setEditingPart] = useState(null)
  const [editForm, setEditForm] = useState({ name: '', code: '', category: '잉크', customer_price: 0, dealer_price: 0, note: '' })

  useEffect(() => {
    fetchCompanies()
    fetchHistory() // 💡 이력 목록 조회 실행
  }, [])

  const fetchCompanies = async () => {
    const { data } = await supabase.from('companies').select('*').order('name', { ascending: true })
    if (data) setCompanies(data)
  }

  // 💡 입·출고증 발행 이력 목록 불러오기 함수
  const fetchHistory = async () => {
    const { data, error } = await supabase
      .from('delivery_history')
      .select('*')
      .order('date', { ascending: false })
      .order('id', { ascending: false })
    if (!error && data) {
      setHistoryList(data)
    }
  }

  const inputStyle = {
    width: '100%', padding: '10px 12px', fontSize: '13px', borderRadius: '10px',
    border: '1px solid #CBD5E1', backgroundColor: '#F8FAFC', outline: 'none', boxSizing: 'border-box'
  }
  const labelStyle = { fontSize: '13px', fontWeight: '700', color: '#475569', marginBottom: '6px', display: 'block' }

  // A. 신규 소모품 품목 등록
  const handleAddPart = async (e) => {
    e.preventDefault()
    if (!partName.trim() || !partCode.trim()) return alert('품명과 품번/코드를 입력해주세요!')

    const calculatedStock = lots.reduce((sum, l) => sum + Number(l.qty || 0), 0)
    const validPrices = companyPrices.filter(cp => cp.company_name.trim() !== '')

    const { error } = await supabase.from('parts').insert([{
      name: partName.trim(),
      code: partCode.trim(),
      category: partCategory,
      dealer_price: Number(dealerPrice),
      customer_price: Number(customerPrice),
      stock: calculatedStock,
      lots: JSON.stringify(lots),
      company_prices: JSON.stringify(validPrices),
      note: partNote,
      is_favorite: false
    }])

    if (!error) {
      alert('신규 소모품이 성공적으로 등록되었습니다.')
      setPartName(''); setPartCode(''); setPartNote(''); setLots([]); setCompanyPrices([]); setShowAddForm(false)
      refreshParts()
    } else {
      alert('등록 실패: ' + error.message)
    }
  }

  // B. 품목 수정 저장
  const handleSaveEdit = async (e) => {
    e.preventDefault()
    if (!editingPart) return

    const { error } = await supabase.from('parts').update({
      name: editForm.name.trim(),
      code: editForm.code.trim(),
      category: editForm.category,
      customer_price: Number(editForm.customer_price),
      dealer_price: Number(editForm.dealer_price),
      note: editForm.note
    }).eq('id', editingPart.id)

    if (!error) {
      alert('품목 정보가 수정되었습니다.')
      setEditingPart(null)
      refreshParts()
    } else {
      alert('수정 실패: ' + error.message)
    }
  }

  // C. 품목 삭제
  const handleDeletePart = async (id, name) => {
    if (!window.confirm(`[${name}] 품목을 정말 삭제하시겠습니까?`)) return
    const { error } = await supabase.from('parts').delete().eq('id', id)
    if (!error) {
      alert('품목이 삭제되었습니다.')
      refreshParts()
    } else {
      alert('삭제 실패: ' + error.message)
    }
  }

  // D. 입고 처리
  const handleQuickAddLot = async (e) => {
    e.preventDefault()
    if (!quickLotPart) return
    if (!quickQty || Number(quickQty) <= 0) return alert('입고 수량을 정확히 입력해주세요!')

    let currentLots = []
    try {
      currentLots = quickLotPart.lots ? (typeof quickLotPart.lots === 'string' ? JSON.parse(quickLotPart.lots) : quickLotPart.lots) : []
    } catch (err) { currentLots = [] }

    const newLot = {
      lot_number: quickLotNo.trim(),
      received_date: quickReceivedDate || new Date().toISOString().split('T')[0],
      expiry: quickExpiry || '',
      qty: Number(quickQty)
    }

    const updatedLots = [...currentLots, newLot]
    const calculatedTotalStock = updatedLots.reduce((sum, l) => sum + Number(l.qty || 0), 0)

    const { error: partError } = await supabase.from('parts').update({
      lots: JSON.stringify(updatedLots),
      stock: calculatedTotalStock
    }).eq('id', quickLotPart.id)

    if (partError) return alert('입고 처리 오류: ' + partError.message)

    const partCodeStr = quickLotPart.code ? `[${quickLotPart.code}] ` : ''
    const summaryText = `${partCodeStr}${quickLotPart.name} (${quickQty}개)`

    await supabase.from('delivery_history').insert([{
      date: quickReceivedDate || new Date().toISOString().split('T')[0],
      company_name: '본사 창고 (입고)',
      equipment_sn: quickLotNo.trim() ? `LOT: ${quickLotNo.trim()}` : '로트 입고',
      summary: summaryText,
      author: '배진철',
      note: quickExpiry ? `유통기한: ${quickExpiry}` : '유통기한 미지정',
      type: 'INBOUND'
    }])

    alert('입고 로트가 추가되고 이력에 등록되었습니다.')
    setQuickLotPart(null); setQuickQty(''); setQuickExpiry(''); setQuickLotNo('');
    refreshParts()
    fetchHistory()
  }

  // E. 로트별 수동 지정 출고 처리 (이력 삭제 시 복원할 수 있도록 상세 로트 정보 숨김 태그 포함 저장)
  const handleOutboundSubmit = async (e) => {
    e.preventDefault()
    if (!outboundPart) return
    if (!outboundCompany.trim()) return alert('공급받는 업체명을 입력해주세요!')

    let currentLots = []
    try {
      currentLots = outboundPart.lots ? (typeof outboundPart.lots === 'string' ? JSON.parse(outboundPart.lots) : outboundPart.lots) : []
    } catch (err) { currentLots = [] }

    let totalOutboundQty = 0
    const deductedDetails = [] // 어떤 로트에서 몇 개가 차감되었는지 추적하는 배열

    try {
      const updatedLots = currentLots.map((lot, idx) => {
        const inputVal = outboundLotInputs[idx]
        const deductQty = inputVal ? Number(inputVal) : 0
        if (deductQty < 0) return lot
        if (deductQty > Number(lot.qty || 0)) {
          alert(`Lot [${lot.lot_number || '미지정'}]의 재고 수량(${lot.qty}개)보다 많은 수량을 출고할 수 없습니다!`)
          throw new Error('Over-deduct')
        }
        if (deductQty > 0) {
          totalOutboundQty += deductQty
          deductedDetails.push({
            lot_number: lot.lot_number || '',
            expiry: lot.expiry || '',
            received_date: lot.received_date || '',
            qty: deductQty
          })
        }
        return {
          ...lot,
          qty: Number(lot.qty || 0) - deductQty
        }
      }).filter(lot => Number(lot.qty) > 0)

      if (totalOutboundQty <= 0) return alert('출고할 수량을 하나 이상 입력해주세요!')

      const calculatedTotalStock = updatedLots.reduce((sum, l) => sum + Number(l.qty || 0), 0)

      const { error: partError } = await supabase.from('parts').update({
        lots: JSON.stringify(updatedLots),
        stock: calculatedTotalStock
      }).eq('id', outboundPart.id)

      if (partError) return alert('출고 처리 오류: ' + partError.message)

      const partCodeStr = outboundPart.code ? `[${outboundPart.code}] ` : ''
      const summaryText = `${partCodeStr}${outboundPart.name} (${totalOutboundQty}개)`
      const lotSummaryStr = deductedDetails.map(d => `${d.lot_number || '미지정'}(${d.qty}개)`).join(', ')
      
      // 💡 삭제 시 정확한 로트로 복원하기 위해 note에 숨김 JSON 태그 포함
      const restoreTag = `[LOT_RESTORE:${JSON.stringify(deductedDetails)}]`
      const finalNote = outboundNote.trim() ? `${outboundNote.trim()} ${restoreTag}` : restoreTag

      await supabase.from('delivery_history').insert([{
        date: outboundDate || new Date().toISOString().split('T')[0],
        company_name: outboundCompany.trim(),
        equipment_sn: lotSummaryStr || '-',
        summary: summaryText,
        author: '배진철',
        note: finalNote,
        type: 'OUTBOUND'
      }])

      alert('출고가 완료되고 이력에 등록되었습니다.')
      setOutboundPart(null); setOutboundCompany(''); setOutboundNote(''); setOutboundLotInputs({})
      refreshParts()
      fetchHistory()
    } catch (err) {
      if (err.message !== 'Over-deduct') console.error(err)
    }
  }

  // 💡 F. 입·출고 이력 삭제 및 재고 자동 복원 함수
  const handleDeleteHistory = async (item) => {
    if (!window.confirm(`[${item.summary}] 내역을 삭제하시겠습니까?\n(연동된 재고와 유통기한 로트 수량이 자동으로 복원됩니다.)`)) return

    // 1. summary에서 품번 코드와 수량 파싱
    const qtyMatch = item.summary?.match(/\((\d+)개\)/)
    const qty = qtyMatch ? Number(qtyMatch[1]) : 0
    const codeMatch = item.summary?.match(/\[(.*?)\]/)
    const partCode = codeMatch ? codeMatch[1] : null

    if (partCode && qty > 0) {
      // 2. parts 테이블에서 해당 부품 조회
      const { data: partData, error: fetchError } = await supabase
        .from('parts')
        .select('*')
        .eq('code', partCode)
        .single()

      if (!fetchError && partData) {
        let currentLots = []
        try {
          currentLots = partData.lots ? (typeof partData.lots === 'string' ? JSON.parse(partData.lots) : partData.lots) : []
        } catch (e) { currentLots = [] }

        if (item.type === 'OUTBOUND') {
          // 💡 출고 이력 삭제 시: 차감되었던 로트별 수량을 원래의 유통기한 로트로 정확히 복원
          let restoredDetails = []
          const restoreTagMatch = item.note?.match(/\[LOT_RESTORE:(.*?)\]/)
          if (restoreTagMatch) {
            try { restoredDetails = JSON.parse(restoreTagMatch[1]) } catch (e) { restoredDetails = [] }
          }

          if (restoredDetails.length > 0) {
            // 저장되어 있던 상세 로트별로 각각 복원
            restoredDetails.forEach(rd => {
              const existingLot = currentLots.find(l => (l.lot_number || '') === rd.lot_number && (l.expiry || '') === rd.expiry)
              if (existingLot) {
                existingLot.qty = Number(existingLot.qty || 0) + rd.qty
              } else {
                currentLots.push({
                  lot_number: rd.lot_number,
                  received_date: rd.received_date || new Date().toISOString().split('T')[0],
                  expiry: rd.expiry,
                  qty: rd.qty
                })
              }
            })
          } else {
            // 백업 태그가 없다면 가장 첫 번째 로트에 복원
            if (currentLots.length > 0) {
              currentLots[0].qty = Number(currentLots[0].qty || 0) + qty
            } else {
              currentLots.push({ lot_number: '복원된로트', received_date: new Date().toISOString().split('T')[0], expiry: '', qty: qty })
            }
          }
        } else if (item.type === 'INBOUND') {
          // 💡 입고 이력 삭제 시: 추가되었던 수량만큼 재고 차감
          let rem = qty
          currentLots = currentLots.map(lot => {
            if (rem <= 0) return lot
            const lQty = Number(lot.qty || 0)
            if (lQty <= rem) {
              rem -= lQty
              return { ...lot, qty: 0 }
            } else {
              const newQ = lQty - rem
              rem = 0
              return { ...lot, qty: newQ }
            }
          }).filter(l => Number(l.qty) > 0)
        }

        const newStock = currentLots.reduce((sum, l) => sum + Number(l.qty || 0), 0)

        // 3. parts 테이블 재고 및 로트 업데이트
        await supabase.from('parts').update({
          lots: JSON.stringify(currentLots),
          stock: newStock
        }).eq('id', partData.id)
      }
    }

    // 4. delivery_history 테이블에서 이력 삭제
    const { error: delError } = await supabase.from('delivery_history').delete().eq('id', item.id)
    if (!delError) {
      alert('이력이 삭제되고 재고가 정상적으로 복원되었습니다.')
      fetchHistory()
      refreshParts()
    } else {
      alert('이력 삭제 실패: ' + delError.message)
    }
  }

  const filteredParts = partsList.filter(p => {
    const cat = p.category || ''
    const isConsumable = cat === '잉크' || cat === '희석제' || cat === '소모품'
    return isConsumable && (!searchTerm.trim() || p.name?.toLowerCase().includes(searchTerm.toLowerCase()) || p.code?.toLowerCase().includes(searchTerm.toLowerCase()))
  }).sort((a, b) => (a.name || '').localeCompare(b.name || '', 'ko-KR'))

  return (
    <div onKeyDown={(e) => { if (e.key === 'Enter') e.preventDefault() }}>
      {showAddForm && (
        <div style={{ backgroundColor: '#ffffff', borderRadius: '16px', padding: '24px', border: '1px solid #E2E8F0', marginBottom: '20px', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.05)' }}>
          <h4 style={{ margin: '0 0 16px 0', fontSize: '16px', fontWeight: '700', color: '#0F172A' }}>🧪 신규 소모품 품목 등록</h4>
          <form onSubmit={handleAddPart} style={{ display: 'flex', flexDirection: 'column', gap: '14px' }}>
            <div style={{ display: 'grid', gridTemplateColumns: '1fr 2fr 1fr', gap: '12px' }}>
              <div><label style={labelStyle}>품번 / 코드*</label><input placeholder="예: 70000-00155" value={partCode} onChange={e => setPartCode(e.target.value)} style={inputStyle} required /></div>
              <div><label style={labelStyle}>품명*</label><input placeholder="예: 잉크 (K-100)" value={partName} onChange={e => setPartName(e.target.value)} style={inputStyle} required /></div>
              <div>
                <label style={labelStyle}>구분</label>
                <select value={partCategory} onChange={e => setPartCategory(e.target.value)} style={inputStyle}>
                  <option value="잉크">잉크</option>
                  <option value="희석제">희석제</option>
                  <option value="소모품">소모품</option>
                </select>
              </div>
            </div>

            <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '12px' }}>
              <div><label style={labelStyle}>소비자 기본가 (원)</label><input type="number" value={customerPrice} onChange={e => setCustomerPrice(e.target.value)} style={inputStyle} /></div>
              <div><label style={labelStyle}>딜러 기본가 (원)</label><input type="number" value={dealerPrice} onChange={e => setDealerPrice(e.target.value)} style={inputStyle} /></div>
            </div>

            <div style={{ borderTop: '1px solid #F1F5F9', paddingTop: '10px' }}>
              <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '8px' }}>
                <span style={{ fontSize: '13px', fontWeight: '700', color: '#1E293B' }}>업체별 개별 특가 (선택)</span>
                <button type="button" onClick={() => setCompanyPrices([...companyPrices, { company_name: '', price: 0 }])} style={{ border: 'none', background: '#EFF6FF', color: '#2563EB', padding: '5px 10px', borderRadius: '6px', fontSize: '11px', fontWeight: '700', cursor: 'pointer' }}>+ 업체 특가 추가</button>
              </div>
              {companyPrices.map((cp, idx) => (
                <div key={idx} style={{ display: 'flex', gap: '8px', alignItems: 'center', marginBottom: '6px' }}>
                  <input type="text" placeholder="업체명..." value={cp.company_name} onChange={e => {
                    const updated = [...companyPrices]; updated[idx].company_name = e.target.value; setCompanyPrices(updated);
                  }} style={{ ...inputStyle, flex: 1.5 }} />
                  <input type="number" placeholder="특가(원)" value={cp.price} onChange={e => {
                    const updated = [...companyPrices]; updated[idx].price = e.target.value; setCompanyPrices(updated);
                  }} style={{ ...inputStyle, flex: 1 }} />
                  <button type="button" onClick={() => setCompanyPrices(companyPrices.filter((_, i) => i !== idx))} style={{ border: 'none', background: 'none', color: '#EF4444', fontSize: '14px', cursor: 'pointer' }}>✕</button>
                </div>
              ))}
            </div>

            <div><label style={labelStyle}>📌 비고</label><textarea rows="2" placeholder="특이사항 입력" value={partNote} onChange={e => setPartNote(e.target.value)} style={inputStyle} /></div>

            <div style={{ display: 'flex', justifyContent: 'flex-end', gap: '8px' }}>
              <button type="button" onClick={() => setShowAddForm(false)} style={{ padding: '10px 18px', background: '#F1F5F9', border: '1px solid #CBD5E1', borderRadius: '8px', cursor: 'pointer', fontWeight: '600' }}>취소</button>
              <button type="submit" style={{ padding: '10px 20px', background: '#2563EB', color: '#fff', border: 'none', borderRadius: '8px', fontWeight: '700', cursor: 'pointer' }}>등록 저장</button>
            </div>
          </form>
        </div>
      )}

      <div style={{ marginBottom: '16px' }}>
        <input type="text" placeholder="🔍 소모품 품명 또는 코드 검색..." value={searchTerm} onChange={e => setSearchTerm(e.target.value)} style={{ ...inputStyle, padding: '12px', maxWidth: '360px' }} />
      </div>

      <div style={{ backgroundColor: '#ffffff', borderRadius: '16px', border: '1px solid #E2E8F0', overflow: 'hidden', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.02)', marginBottom: '30px' }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left', fontSize: '14px', tableLayout: 'fixed' }}>
          <thead>
            <tr style={{ backgroundColor: '#F8FAFC', borderBottom: '1px solid #E2E8F0', color: '#475569' }}>
              <th style={{ padding: '14px 16px', width: '80px' }}>구분</th>
              <th style={{ padding: '14px 16px', width: '120px' }}>품번 / 코드</th>
              <th style={{ padding: '14px 16px', width: '180px' }}>품명</th>
              <th style={{ padding: '14px 16px', width: '160px' }}>재고 수량 및 유통기한</th>
              <th style={{ padding: '14px 16px', width: '140px' }}>업체별 특가</th>
              <th style={{ padding: '14px 16px', width: '100px' }}>소비자가</th>
              <th style={{ padding: '14px 16px', width: '100px' }}>딜러가</th>
              <th style={{ padding: '14px 16px', width: '160px', textAlign: 'center' }}>재고 입·출고</th>
              <th style={{ padding: '14px 16px', width: '110px', textAlign: 'center' }}>관리 (수정/삭제)</th>
            </tr>
          </thead>
          <tbody>
            {filteredParts.length === 0 ? (
              <tr><td colSpan="9" style={{ padding: '40px', textAlign: 'center', color: '#94A3B8' }}>등록된 소모품이 없습니다.</td></tr>
            ) : (
              filteredParts.map(p => {
                let parsedCompanyPrices = []
                try {
                  const temp = p.company_prices ? (typeof p.company_prices === 'string' ? JSON.parse(p.company_prices) : p.company_prices) : []

                  {(!outboundPart.lots || (typeof outboundPart.lots === 'string' ? JSON.parse(outboundPart.lots) : outboundPart.lots).length === 0) ? (

              </div>
            </form>

      {selectedExpiryPart && (

import { supabase } from '../../supabaseClient'

export default function DeliveryHistoryTab({ refreshParts }) {
  const [historyList, setHistoryList] = useState([])
  const [searchTerm, setSearchTerm] = useState('')
  const [filterType, setFilterType] = useState('ALL') // ALL, INBOUND, OUTBOUND
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetchHistory()
  }, [])

  const fetchHistory = async () => {
    setLoading(true)
    const { data, error } = await supabase
      .from('delivery_history')
      .select('*')
      .order('date', { ascending: false })
      .order('id', { ascending: false })

    if (!error && data) {
      setHistoryList(data)
    }
    setLoading(false)
  }

  // 💡 이력 삭제 및 재고(로트/S/N) 자동 복원 함수
  const handleDeleteHistory = async (item) => {
    if (!window.confirm(`[${item.summary}]\n이 내역을 삭제하시겠습니까?\n(삭제 시 연동된 재고 수량 및 로트/S/N가 자동으로 복원됩니다.)`)) return

    // 1. summary에서 품번 코드와 수량 추출 (예: "[70000-00122] 잉크 (10개)")
    const qtyMatch = item.summary?.match(/\((\d+)개\)/)
    const qty = qtyMatch ? Number(qtyMatch[1]) : 0
    const codeMatch = item.summary?.match(/\[(.*?)\]/)
    const partCode = codeMatch ? codeMatch[1] : null

    if (partCode && qty > 0) {
      // 2. parts 테이블에서 해당 부품 조회 (같은 코드가 여러 구분으로 있을 수 있으므로 배열로 조회)
      const { data: partsMatched, error: fetchError } = await supabase
        .from('parts')
        .select('*')
        .eq('code', partCode)

      if (!fetchError && partsMatched && partsMatched.length > 0) {
        // 보통 첫 번째 매칭된 품목 대상 (구분이 다를 경우 첫 번째 항목)
        const partData = partsMatched[0]
        const cat = partData.category || ''
        const isConsumable = cat === '잉크' || cat === '희석제' || cat === '소모품'

        if (isConsumable) {
          // 🧪 소모품 (로트/유통기한 기반 복원)
          let currentLots = []
          try {
            currentLots = partData.lots ? (typeof partData.lots === 'string' ? JSON.parse(partData.lots) : partData.lots) : []
          } catch (e) { currentLots = [] }

          if (item.type === 'OUTBOUND') {
            // 출고 삭제 -> 차감되었던 로트별 수량 원상복구
            let restoredDetails = []
            const restoreTagMatch = item.note?.match(/\[LOT_RESTORE:(.*?)\]/)
            if (restoreTagMatch) {
              try { restoredDetails = JSON.parse(restoreTagMatch[1]) } catch (e) { restoredDetails = [] }
            }

            if (restoredDetails.length > 0) {
              restoredDetails.forEach(rd => {
                const existingLot = currentLots.find(l => (l.lot_number || '') === rd.lot_number && (l.expiry || '') === rd.expiry)
                if (existingLot) {
                  existingLot.qty = Number(existingLot.qty || 0) + Number(rd.qty || 0)
                } else {
                  currentLots.push({
                    lot_number: rd.lot_number,
                    received_date: rd.received_date || new Date().toISOString().split('T')[0],
                    expiry: rd.expiry,
                    qty: Number(rd.qty || 0)
                  })
                }
              })
            } else {
              // 상세 태그가 없는 구버전 출고 이력일 경우 첫 번째 로트에 합산
              if (currentLots.length > 0) {
                currentLots[0].qty = Number(currentLots[0].qty || 0) + qty
              } else {
                currentLots.push({ lot_number: '복원재고', received_date: new Date().toISOString().split('T')[0], expiry: '', qty: qty })
              }
            }
          } else if (item.type === 'INBOUND') {
            // 입고 삭제 -> 추가되었던 수량만큼 차감
            let rem = qty
            currentLots = currentLots.map(lot => {
              if (rem <= 0) return lot
              const lQty = Number(lot.qty || 0)
              if (lQty <= rem) {
                rem -= lQty
                return { ...lot, qty: 0 }
              } else {
                const newQ = lQty - rem
                rem = 0
                return { ...lot, qty: newQ }
              }
            }).filter(l => Number(l.qty) > 0)
          }

          const newStock = currentLots.reduce((sum, l) => sum + Number(l.qty || 0), 0)
          await supabase.from('parts').update({
            lots: JSON.stringify(currentLots),
            stock: newStock
          }).eq('id', partData.id)

        } else {
          // ⚙️ 일반 부품 (S/N 기반 복원)
          let currentSerials = []
          try {
            currentSerials = partData.serials ? (typeof partData.serials === 'string' ? JSON.parse(partData.serials) : partData.serials) : []
          } catch (e) { currentSerials = [] }

          // equipment_sn에서 S/N 추출 (예: "S/N: LJ-3516 (BG)")
          const snMatch = item.equipment_sn?.match(/S\/N:\s*([^\s(]+)/)
          const snValue = snMatch ? snMatch[1].trim() : null
          const condMatch = item.equipment_sn?.match(/\((BG|EX|OH)\)/)
          const condValue = condMatch ? condMatch[1] : 'BG'

          if (item.type === 'INBOUND' && snValue) {
            // 입고 삭제 -> 해당 S/N 제거
            currentSerials = currentSerials.filter(s => s.sn !== snValue)
          } else if (item.type === 'OUTBOUND' && snValue) {
            // 출고 삭제 -> 해당 S/N 다시 복원
            if (!currentSerials.some(s => s.sn === snValue)) {
              currentSerials.push({ sn: snValue, condition: condValue, received_date: item.date })
            }
          }

          await supabase.from('parts').update({
            serials: JSON.stringify(currentSerials),
            stock: currentSerials.length
          }).eq('id', partData.id)
        }
      }
    }

    // 3. delivery_history 테이블에서 이력 행 삭제
    const { error: delError } = await supabase.from('delivery_history').delete().eq('id', item.id)
    if (!delError) {
      alert('이력이 삭제되고 재고가 정상적으로 복원되었습니다.')
      fetchHistory()
      if (typeof refreshParts === 'function') refreshParts()
    } else {
      alert('이력 삭제 실패: ' + delError.message)
    }
  }

  // 화면에 표시할 때 숨김 복원 태그([LOT_RESTORE:...])는 깔끔하게 가려서 출력
  const formatCleanNote = (noteText) => {
    if (!noteText) return '-'
    const cleaned = noteText.replace(/\[LOT_RESTORE:.*?\]/g, '').trim()
    return cleaned || '-'
  }

  const filteredHistory = historyList.filter(item => {
    if (filterType !== 'ALL' && item.type !== filterType) return false
    if (!searchTerm.trim()) return true
    const q = searchTerm.toLowerCase()
    return (
      item.company_name?.toLowerCase().includes(q) ||
      item.summary?.toLowerCase().includes(q) ||
      item.equipment_sn?.toLowerCase().includes(q)
    )
  })

  return (
    <div>
      {/* 상단 필터 및 검색 바 */}
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '16px', flexWrap: 'wrap', gap: '10px' }}>
        <div style={{ display: 'flex', gap: '8px' }}>
          {[
            { id: 'ALL', label: '전체 이력' },
            { id: 'OUTBOUND', label: '📤 출고 내역' },
            { id: 'INBOUND', label: '📥 입고 내역' }
          ].map(tab => (
            <button
              key={tab.id}
              type="button"
              onClick={() => setFilterType(tab.id)}
              style={{
                padding: '8px 14px',
                borderRadius: '10px',
                border: filterType === tab.id ? '1px solid #2563EB' : '1px solid #CBD5E1',
                backgroundColor: filterType === tab.id ? '#EFF6FF' : '#ffffff',
                color: filterType === tab.id ? '#2563EB' : '#475569',
                fontWeight: '700',
                fontSize: '13px',
                cursor: 'pointer'
              }}
            >
              {tab.label}
            </button>
          ))}
        </div>

        <input
          type="text"
          placeholder="🔍 거래처, 품명, 코드, Lot/SN 검색..."
          value={searchTerm}
          onChange={e => setSearchTerm(e.target.value)}
          style={{
            padding: '10px 14px',
            fontSize: '13px',
            borderRadius: '10px',
            border: '1px solid #CBD5E1',
            backgroundColor: '#F8FAFC',
            width: '300px',
            outline: 'none'
          }}
        />
      </div>

      {/* 이력 테이블 */}
      <div style={{ backgroundColor: '#ffffff', borderRadius: '16px', border: '1px solid #E2E8F0', overflow: 'hidden', boxShadow: '0 4px 6px -1px rgba(0,0,0,0.02)' }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left', fontSize: '13px' }}>
          <thead>
            <tr style={{ backgroundColor: '#F8FAFC', borderBottom: '1px solid #E2E8F0', color: '#475569' }}>
              <th style={{ padding: '14px 16px', width: '80px', textAlign: 'center' }}>구분</th>
              <th style={{ padding: '14px 16px', width: '110px' }}>일자</th>
              <th style={{ padding: '14px 16px', width: '160px' }}>거래처 / 구분</th>
              <th style={{ padding: '14px 16px' }}>품명 및 수량 요약</th>
              <th style={{ padding: '14px 16px', width: '160px' }}>Lot / S/N 내역</th>
              <th style={{ padding: '14px 16px', width: '160px' }}>비고</th>
              <th style={{ padding: '14px 16px', width: '80px', textAlign: 'center' }}>작성자</th>
              <th style={{ padding: '14px 16px', width: '110px', textAlign: 'center' }}>관리</th>
            </tr>
          </thead>
          <tbody>
            {loading ? (
              <tr><td colSpan="8" style={{ padding: '40px', textAlign: 'center', color: '#94A3B8' }}>이력을 불러오는 중입니다...</td></tr>
            ) : filteredHistory.length === 0 ? (
              <tr><td colSpan="8" style={{ padding: '40px', textAlign: 'center', color: '#94A3B8' }}>조회된 입·출고 이력이 없습니다.</td></tr>
            ) : (
              filteredHistory.map(item => (
                <tr key={item.id} style={{ borderBottom: '1px solid #F1F5F9', verticalAlign: 'middle' }}>
                  <td style={{ padding: '12px 16px', textAlign: 'center' }}>
                    <span style={{
                      padding: '4px 8px',
                      borderRadius: '6px',
                      fontSize: '11px',
                      fontWeight: '800',
                      backgroundColor: item.type === 'OUTBOUND' ? '#FEF3C7' : '#DEF7EC',
                      color: item.type === 'OUTBOUND' ? '#D97706' : '#03543F'
                    }}>
                      {item.type === 'OUTBOUND' ? '출고' : '입고'}
                    </span>
                  </td>
                  <td style={{ padding: '12px 16px', color: '#475569', fontWeight: '600' }}>{item.date}</td>
                  <td style={{ padding: '12px 16px', fontWeight: '700', color: '#0F172A' }}>{item.company_name}</td>
                  <td style={{ padding: '12px 16px', fontWeight: '700', color: '#1E293B' }}>{item.summary}</td>
                  <td style={{ padding: '12px 16px', color: '#2563EB', fontWeight: '600', fontSize: '12px' }}>{item.equipment_sn || '-'}</td>
                  <td style={{ padding: '12px 16px', color: '#64748B', fontSize: '12px' }}>{formatCleanNote(item.note)}</td>
                  <td style={{ padding: '12px 16px', textAlign: 'center', color: '#475569' }}>{item.author || '-'}</td>
                  <td style={{ padding: '12px 16px', textAlign: 'center' }}>
                    <button
                      type="button"
                      onClick={() => handleDeleteHistory(item)}
                      style={{
                        background: '#FEE2E2',
                        color: '#EF4444',
                        border: 'none',
                        padding: '5px 10px',
                        borderRadius: '6px',
                        fontSize: '11px',
                        fontWeight: '700',
                        cursor: 'pointer'
                      }}
                    >
                      삭제(복원)
                    </button>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>
    </div>
  )
}
